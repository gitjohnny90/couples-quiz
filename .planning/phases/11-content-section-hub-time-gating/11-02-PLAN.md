    handlePickSection function (GATE-03 — fresh-fetch-before-write pattern):
    ```
    async function handlePickSection(sectionId) {
      // 1. Fresh fetch to avoid race condition
      const { data: fresh } = await supabase
        .from('responses')
        .select('answers')
        .eq('session_id', sessionId)
        .eq('pack_id', 'daily-photo-challenge')
        .eq('player_id', 'shared')
        .maybeSingle()
      const currentState = fresh?.answers ?? { completedSections: {}, inProgressSectionId: null, lastCompletedAt: null }

      // 2. Guard: don't overwrite if frozen or already in progress
      if (isGloballyFrozen(currentState)) return
      if (currentState.inProgressSectionId) return

      // 3. Write new state
      const newState = { ...currentState, inProgressSectionId: sectionId }
      await supabase
        .from('responses')
        .upsert({
          session_id: sessionId,
          pack_id: 'daily-photo-challenge',
          player_id: 'shared',
          answers: newState,
        }, { onConflict: 'session_id,pack_id,player_id' })

      // 4. Navigate to prompt flow (Phase 12 route — will 404 for now, that's expected)
      navigate(`/daily-photo-section/${sessionId}/${sectionId}`)
    }
    ```
